cmake_minimum_required(VERSION 3.31)
project(AccessibilityProbe LANGUAGES CXX OBJCXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_OBJCXX_STANDARD 17)
find_package(Qt6 6.5.3 EXACT REQUIRED COMPONENTS Widgets Test)
get_filename_component(MEDIAMUSTER_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
add_executable(accessibility_probe probe.mm "${MEDIAMUSTER_ROOT}/src/qtaccessibilityfix.cpp")
target_include_directories(accessibility_probe PRIVATE "${MEDIAMUSTER_ROOT}/src")
target_link_libraries(accessibility_probe PRIVATE Qt6::Widgets Qt6::Test "-framework AppKit")
